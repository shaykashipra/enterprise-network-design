! Device: CORE-HQ
! Model: Cisco 3560
! Purpose: Layer-3 core switch, VLAN gateways, DHCP, OSPF, ACL

enable
configure terminal
hostname CORE-HQ
ip routing

vlan 10
 name HR
vlan 20
 name FINANCE
vlan 30
 name IT
vlan 40
 name SALES
vlan 50
 name MGMT
vlan 60
 name SERVERS

interface fa0/1
 description TRUNK_TO_ASW-HQ
 switchport trunk encapsulation dot1q
 switchport mode trunk

interface vlan 10
 description HR_GATEWAY
 ip address 10.10.10.1 255.255.255.0
 ip access-group HR_BLOCK_FINANCE in
 no shutdown

interface vlan 20
 description FINANCE_GATEWAY
 ip address 10.10.20.1 255.255.255.0
 no shutdown

interface vlan 30
 description IT_GATEWAY
 ip address 10.10.30.1 255.255.255.0
 no shutdown

interface vlan 40
 description SALES_GATEWAY
 ip address 10.10.40.1 255.255.255.0
 no shutdown

interface vlan 50
 description MGMT_GATEWAY
 ip address 10.10.50.1 255.255.255.0
 no shutdown

interface vlan 60
 description SERVER_GATEWAY
 ip address 10.10.60.1 255.255.255.0
 no shutdown

interface gi0/1
 description ROUTED_LINK_TO_R-HQ
 no switchport
 ip address 10.10.99.1 255.255.255.252
 no shutdown

ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.30.1 10.10.30.20
ip dhcp excluded-address 10.10.40.1 10.10.40.20
ip dhcp excluded-address 10.10.50.1 10.10.50.20

ip dhcp pool HR
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 8.8.8.8

ip dhcp pool FINANCE
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 8.8.8.8

ip dhcp pool IT
 network 10.10.30.0 255.255.255.0
 default-router 10.10.30.1
 dns-server 8.8.8.8

ip dhcp pool SALES
 network 10.10.40.0 255.255.255.0
 default-router 10.10.40.1
 dns-server 8.8.8.8

ip access-list extended HR_BLOCK_FINANCE
 deny ip 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255
 permit ip any any

router ospf 1
 router-id 1.1.1.1
 network 10.10.10.0 0.0.0.255 area 0
 network 10.10.20.0 0.0.0.255 area 0
 network 10.10.30.0 0.0.0.255 area 0
 network 10.10.40.0 0.0.0.255 area 0
 network 10.10.50.0 0.0.0.255 area 0
 network 10.10.60.0 0.0.0.255 area 0
 network 10.10.99.0 0.0.0.3 area 0

end
copy running-config startup-config
